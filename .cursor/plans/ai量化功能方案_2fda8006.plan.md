## 业务口径确认（与你描述一致）

以下为对需求的核对结论，实现时以本节为产品真值。

1. **用户投入**：用户用现货 **1000 USDT** 发起 AI 量化认购 → 现货扣款 → 生成 `app_ai_quant_subscription`，`audit_status=0` 待审。
2. **后台审核**：管理员审核 **通过** 后 `audit_status=1`，此时才允许为该用户（及对应认购单）在 manage **创建**量化展示订单。
3. **后台建单**：管理员录入例如 **项目名称「BTC」**、`buy_price=70000`（表示 **70000 USDT 买 1 BTC** 的单价）、`buy_usdt=1000`（本次用多少 USDT 参与该笔展示单的「本金」口径，与认购 1000 对应由运营保证一致）。
4. **折算买入数量（币本位）**：`coin_quantity = buy_usdt / buy_price`（例：`1000 / 70000 ≈ 0.0142857` BTC）。该值可 **发布时写入** `coin_quantity` 冗余列便于列表展示，也可仅接口计算不落库。
5. **卖出与盈亏（USDT）**：设 `sell_price=72000`，则 **盈亏 USDT** = `(sell_price - buy_price) * coin_quantity` = `(72000 - 70000) * 0.0142857 ≈ 28.57`（你举例 28.4 为四舍五入口径差异，实现用 `BigDecimal` 指定精度）。
6. **等价公式（实现可任选其一，结果一致）**：`profit_amount = (sell_price - buy_price) / buy_price * buy_usdt`。
7. **盈亏百分比（相对本金的小数）**：`profit_rate = profit_amount / buy_usdt`（例：`28.4 / 1000 = 0.0284`）。**注意**：这不是 `* 100` 后的「2.84」百分数；若前端要显示 **2.84%**，对 `profit_rate` 乘 100 或单独增加展示字段（计划内约定写进实体/接口中文注释）。
8. **用户可见**：草稿阶段 `user_visible=0`，用户 **看不到**；管理员在卖出价、卖出时间等填完后调用 **发布**（计划中为 `/order/publish`），置 `user_visible=1`，即你所说的 **「再手动让用户能看到这笔交易」**。

**与旧计划片段的差异**：原示例代码中 `profit_rate = profit_amount / buy_usdt * 100` 已改为 **不乘 100**，以符合「0.0284」口径。

import org.lemon.api.modules.aiquant.service.AppAiQuantSubscriptionService
import org.lemon.api.modules.appuser.service.AppUserService
import org.lemon.api.modules.userwallet.domain.dto.ChangeReq
import org.lemon.api.modules.userwallet.enums.GoldChangeEnum
import org.lemon.api.modules.userwallet.service.AppUserWalletService
import org.springframework.stereotype.Service
import java.math.BigDecimal

@Service
open class AppAiQuantSubscriptionServiceImpl(
    private val walletService: AppUserWalletService,
    private val userService: AppUserService,
) : ServiceImpl<AppAiQuantSubscriptionMapper, AppAiQuantSubscription>(),
    AppAiQuantSubscriptionService {

    private val logger = KotlinLogging.logger {}

    override fun subscribe(userId: Long, amount: BigDecimal) {
        val key = RedisKeys.LOCK_AI_QUANT_SUBSCRIBE + userId
        RedisLockService.lockTransaction(key) {
            val orderNo = "AQ" + GeneratorIdUtil.generateId()

            // 从现货(USDT)钱包扣款；须传 checkBalance=true（与理财购买一致）余额不足抛 insufficient_wallet_balance
            walletService.balanceChange(
                ChangeReq.build(userId, Constants.USDT, GoldChangeEnum.AI_QUANT_SUBSCRIBE) {
                    balance = amount.negate()
                    remark = "AI量化认购扣款,单号[$orderNo]"
                },
                true
            )

            // 冗余用户信息写入认购记录
            val appUser = userService.getById(userId)
            val sub = AppAiQuantSubscription()
            sub.userId = appUser.id
            sub.uid = appUser.uid
            sub.userName = appUser.userName
            sub.userGroup = appUser.userGroup
            sub.topUserId = appUser.topUserId
            sub.t2Id = appUser.t2Id
            sub.orderNo = orderNo
            sub.coinId = Constants.USDT
            sub.amount = amount
            sub.auditStatus = 0           // 待审核
            sub.subscriptionStatus = 0    // 正常
            save(sub)
            logger.info { "AI量化认购成功: userId=$userId, amount=$amount, orderNo=$orderNo" }
        }
    }

    override fun redeem(userId: Long, subscriptionId: Long) {
        val key = RedisKeys.LOCK_AI_QUANT_REDEEM + userId
        RedisLockService.lockTransaction(key) {
            val sub = getById(subscriptionId)
                ?: throw BusinessException("params_error")
            // 校验: 必须是当前用户的、审核通过的、未赎回的认购单
            if (sub.userId != userId) throw BusinessException("params_error")
            if (sub.auditStatus != 1) throw BusinessException("ai_quant_not_approved")
            if (sub.subscriptionStatus != 0) throw BusinessException("ai_quant_already_redeemed")

            // 退款到现货钱包
            walletService.forceBalanceChange(
                ChangeReq.build(userId, Constants.USDT, GoldChangeEnum.AI_QUANT_REDEEM) {
                    balance = sub.amount
                    remark = "AI量化赎回退款,单号[${sub.orderNo}]"
                }
            )

            // 更新认购状态为已赎回
            update(
                KtUpdateWrapper(AppAiQuantSubscription())
                    .eq(AppAiQuantSubscription::id, subscriptionId)
                    .set(AppAiQuantSubscription::subscriptionStatus, 1)
                    .set(AppAiQuantSubscription::redeemTime, DateUtil.date())
            )
            logger.info { "AI量化赎回成功: userId=$userId, subscriptionId=$subscriptionId" }
        }
    }

    override fun audit(adminId: Long, subscriptionId: Long, pass: Boolean, reason: String?) {
        val sub = getById(subscriptionId)
            ?: throw BusinessException("params_error")
        if (sub.auditStatus != 0) {
            throw BusinessException("ai_quant_already_audited")
        }

        if (pass) {
            // 审核通过：仅更新状态
            update(
                KtUpdateWrapper(AppAiQuantSubscription())
                    .eq(AppAiQuantSubscription::id, subscriptionId)
                    .set(AppAiQuantSubscription::auditStatus, 1)
                    .set(AppAiQuantSubscription::auditAdminId, adminId)
                    .set(AppAiQuantSubscription::auditTime, DateUtil.date())
            )
        } else {
            // 审核驳回：退款 + 更新状态
            walletService.forceBalanceChange(
                ChangeReq.build(sub.userId!!, Constants.USDT, GoldChangeEnum.AI_QUANT_REJECT_REFUND) {
                    balance = sub.amount
                    remark = "AI量化认购驳回退款,单号[${sub.orderNo}]"
                }
            )
            update(
                KtUpdateWrapper(AppAiQuantSubscription())
                    .eq(AppAiQuantSubscription::id, subscriptionId)
                    .set(AppAiQuantSubscription::auditStatus, -1)
                    .set(AppAiQuantSubscription::auditAdminId, adminId)
                    .set(AppAiQuantSubscription::auditTime, DateUtil.date())
                    .set(AppAiQuantSubscription::rejectReason, reason)
            )
        }
        logger.info { "AI量化审核: subscriptionId=$subscriptionId, pass=$pass, adminId=$adminId" }
    }
}

@Service
open class AppAiQuantOrderServiceImpl(
    private val subscriptionService: AppAiQuantSubscriptionService,

    private val logger = KotlinLogging.logger {}

    override fun createOrder(adminId: Long, order: AppAiQuantOrder) {
        // 校验关联的认购单: 必须存在且审核通过
        val sub = subscriptionService.getById(order.subscriptionId)
            ?: throw BusinessException("params_error")
        if (sub.auditStatus != 1) {
            throw BusinessException("ai_quant_subscription_not_approved")
        }
        // 校验用户一致性
        if (sub.userId != order.userId) {
            throw BusinessException("ai_quant_user_mismatch")
        }

        // 自动冗余用户信息（从认购单复制，保证数据一致）
        order.uid = sub.uid
        order.userName = sub.userName
        order.userGroup = sub.userGroup
        order.topUserId = sub.topUserId
        order.t2Id = sub.t2Id
        order.orderNo = "AQO" + GeneratorIdUtil.generateId()
        order.settleCoin = order.settleCoin ?: Constants.USDT
        order.userVisible = 0   // 创建时默认不可见
        order.createAdminId = adminId
        save(order)
        logger.info { "AI量化订单创建: orderId=${order.id}, subscriptionId=${order.subscriptionId}" }
    }

    override fun updateOrder(adminId: Long, order: AppAiQuantOrder) {
        val existing = getById(order.id)
            ?: throw BusinessException("params_error")
        // 已发布的订单不可修改
        if (existing.userVisible == 1) {
            throw BusinessException("ai_quant_order_published")
        }
        updateById(order)
        logger.info { "AI量化订单更新: orderId=${order.id}, adminId=$adminId" }
    }

    override fun publish(adminId: Long, orderId: Long) {
        val order = getById(orderId)
            ?: throw BusinessException("params_error")
        if (order.userVisible == 1) {
            throw BusinessException("ai_quant_order_already_published")
        }
        // 发布前必须有卖出时间和卖出价格
        if (order.sellTime == null || order.sellPrice == null) {
            throw BusinessException("ai_quant_order_incomplete")
        }

        // 自动计算盈亏（管理员未手动填入 profitAmount 时）
        if (order.profitAmount == null) {
            val buyPrice = order.buyPrice!!
            val sellPrice = order.sellPrice!!
            val buyUsdt = order.buyUsdt!!
            if (buyPrice > BigDecimal.ZERO) {
                // 与手算一致: (卖出价-买入价)/买入价*本金USDT = (卖出价-买入价)*币数量
                order.profitAmount = sellPrice.subtract(buyPrice)
                    .divide(buyPrice, 16, RoundingMode.HALF_UP)
                    .multiply(buyUsdt)
            }
        }
        // 买入数量冗余，便于后台/前端直接展示「买了多少币」
        val buyUsdt = order.buyUsdt!!
        val buyPrice = order.buyPrice!!
        if (buyPrice > BigDecimal.ZERO) {
            order.coinQuantity = buyUsdt.divide(buyPrice, 16, RoundingMode.HALF_UP)
        }
        if (order.profitRate == null && order.profitAmount != null) {
            if (buyUsdt > BigDecimal.ZERO) {
                // 相对本金的收益率小数，如 0.0284（不要在此处 *100）
                order.profitRate = order.profitAmount!!
                    .divide(buyUsdt, 16, RoundingMode.HALF_UP)
            }
        }

        // 设为用户可见（对应「手动让用户能看到这笔交易」）
        update(
            KtUpdateWrapper(AppAiQuantOrder())
                .eq(AppAiQuantOrder::id, orderId)
                .set(AppAiQuantOrder::userVisible, 1)
                .set(AppAiQuantOrder::profitAmount, order.profitAmount)
                .set(AppAiQuantOrder::profitRate, order.profitRate)
                .set(AppAiQuantOrder::coinQuantity, order.coinQuantity)
        )
        logger.info { "AI量化订单发布: orderId=$orderId, profit=${order.profitAmount}" }
    }
}

                        2 -> eq(AppAiQuantOrder::t2Id, adminId)
                    }
                }
                .eq(req.uid != null, AppAiQuantOrder::uid, req.uid)
                .eq(StrUtil.isNotBlank(req.userName), AppAiQuantOrder::userName, req.userName)
                .eq(req.userVisible != null, AppAiQuantOrder::userVisible, req.userVisible)
                .eq(req.userGroup != null, AppAiQuantOrder::userGroup, req.userGroup)
                .eq(req.subscriptionId != null, AppAiQuantOrder::subscriptionId, req.subscriptionId)
                .orderByDesc(AppAiQuantOrder::createTime)
        )
        return R.success(page)
    }
